## Problem

PR-326 declared WorkInhibitor type. To use it, daemon needs a function that walks current Redis state and produces a list of currently-active inhibitors. Doing this in PR-329 alongside dispatcher migration mixes concerns. This PR isolates the derivation logic with no behavior change.

## Solution

Add `derive_active_inhibitors(state: RepoState, redis: Redis, cfg: DaemonConfig) -> list[WorkInhibitor]` in `src/inhibitor.py`. Reads each of the 8 sources and constructs zero or more WorkInhibitor instances. Returns empty list when nothing inhibits. NO callers wire it yet; PR-328 adds RepoState field, PR-329 starts consumption.

### Function skeleton

```python
async def derive_active_inhibitors(
    state: "RepoState",
    redis: "Redis",
    cfg: "DaemonConfig",
    now: Optional[datetime] = None,
) -> list[WorkInhibitor]:
    """Walk current Redis state, return typed list of active inhibitors.

    Read-only derivation. Does NOT change any state.
    """
    from datetime import datetime as _dt, timezone
    current = now if now is not None else _dt.now(timezone.utc)
    inhibitors: list[WorkInhibitor] = []

    if state.user_paused:
        inhibitors.append(WorkInhibitor(
            inhibitor_type=InhibitorType.USER_PAUSE,
            reason_text="Operator paused",
            source_key=f"state:{state.name}.user_paused",
        ))

    stop_ttl = await redis.ttl(f"control:{state.name}:stop")
    if stop_ttl > 0:
        inhibitors.append(WorkInhibitor(
            inhibitor_type=InhibitorType.USER_STOP,
            expires_at=current + timedelta(seconds=stop_ttl),
            reason_text="Operator requested stop",
            source_key=f"control:{state.name}:stop",
        ))

    if state.rate_limited_until_by_coder:
        for coder, until in state.rate_limited_until_by_coder.items():
            if until > current:
                inhibitors.append(WorkInhibitor(
                    inhibitor_type=InhibitorType.RATE_LIMIT,
                    coder_affected=coder,
                    expires_at=until,
                    reason_text=f"{coder} rate-limited",
                    source_key=f"state:{state.name}.rate_limited_until_by_coder.{coder}",
                ))

    if state.spend_ceiling_session_pct >= 100.0 or state.spend_ceiling_weekly_pct >= 100.0:
        inhibitors.append(WorkInhibitor(
            inhibitor_type=InhibitorType.SPEND_CEILING,
            reason_text=f"Spend ceiling reached (session={state.spend_ceiling_session_pct:.0f}%, weekly={state.spend_ceiling_weekly_pct:.0f}%)",
            source_key=f"state:{state.name}.spend_ceiling_*_pct",
        ))

    github_pct = await _read_github_budget_percent(redis)
    if github_pct < cfg.github_api_pause_threshold_percent:
        inhibitors.append(WorkInhibitor(
            inhibitor_type=InhibitorType.GITHUB_BUDGET_PAUSE,
            reason_text=f"GitHub budget at {github_pct:.0f}%",
            source_key="github_rate_limit_budget_*",
        ))
    elif github_pct < cfg.github_api_slowdown_threshold_percent:
        inhibitors.append(WorkInhibitor(
            inhibitor_type=InhibitorType.GITHUB_BUDGET_SLOWDOWN,
            reason_text=f"GitHub budget at {github_pct:.0f}%, slowdown active",
            source_key="github_rate_limit_budget_*",
        ))

    if await redis.exists(f"cascade_panic:{state.name}"):
        inhibitors.append(WorkInhibitor(
            inhibitor_type=InhibitorType.CASCADE_PANIC,
            reason_text="Cascade panic mode auto-stop",
            source_key=f"cascade_panic:{state.name}",
        ))

    if state.error_rate_auto_paused_at is not None:
        inhibitors.append(WorkInhibitor(
            inhibitor_type=InhibitorType.ERROR_RATE_AUTO_PAUSE,
            reason_text="Auto-paused due to ERROR rate threshold",
            source_key=f"state:{state.name}.error_rate_auto_paused_at",
        ))

    return inhibitors
```
